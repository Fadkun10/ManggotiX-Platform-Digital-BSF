import math

# Model Matematika Biologi Dekomposisi Sampah
def calculate_bsf_yield(sampah_kg, populasi_larva):
    laju_makan_maggot = 0.15 # Kg sampah terurai per larva selama siklus hidup
    biokonversi_efisiensi = 0.20 # 20% berat terkonversi menjadi maggot
    rasio_pupuk = 0.40 # 40% sisa pakan menjadi pupuk frass

    kapasitas_maksimal_makan = populasi_larva * laju_makan_maggot
    sampah_terolah = min(sampah_kg, kapasitas_maksimal_makan)

    maggot_wet = sampah_terolah * biokonversi_efisiensi
    maggot_dry = maggot_wet * 0.32 # Kadar kering 32%
    frass_pupuk = sampah_terolah * rasio_pupuk
    reduksi_co2 = sampah_terolah * 2.5 # Faktor reduksi gas metana/CO2

    return {
        "maggot_wet_kg": round(maggot_wet, 2),
        "maggot_dry_kg": round(maggot_dry, 2),
        "frass_kg": round(frass_pupuk, 2),
        "co2_kg": round(reduksi_co2, 2)
    }
